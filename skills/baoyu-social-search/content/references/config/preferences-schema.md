# 搜索偏好设置 schema

```yaml
# 默认平台
default_platform: both  # xhs / zhihu / both

# 默认排序
default_sort: hot  # hot / latest / top

# 默认结果数
default_limit: 10

# 默认时间范围（天）
default_days: 30

# 报告详细程度
report_verbosity: detailed  # brief / detailed / full

# 是否自动下载图片
download_images: false

# 偏好语言
language: zh  # zh / en
```
